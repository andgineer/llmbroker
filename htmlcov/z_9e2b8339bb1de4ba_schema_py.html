106async def ensure_schema(pool: asyncpg.Pool) -> None: 

107 """Create the package's tables/indexes if missing. Idempotent, version-aware. 

108 

109 Migrates inside one transaction; concurrent callers serialize on 

110 postgres's own ``ACCESS EXCLUSIVE`` lock for `CREATE`/`ALTER TABLE`, so 

111 (unlike sqlite) no separate cross-process lock is needed here. 

112 """ 

113 if id(pool) in _schema_ready: 

114 return 

115 async with _schema_lock: 

116 if id(pool) in _schema_ready: 

117 return 

118 async with pool.acquire() as conn, conn.transaction(): 

119 await conn.execute(_CREATE_VERSION_TABLE) 

120 row = await conn.fetchrow("SELECT version FROM llmbroker_schema_version WHERE id = 1") 

121 current = int(row["version"]) if row else 0 

122 if current < _SCHEMA_VERSION: 

123 await conn.execute(_DROP_STATE) 

124 await conn.execute(_DROP_SUMMARIES) 

125 await conn.execute(_DDL) 

126 if current < _SCHEMA_VERSION: 

127 await conn.execute(_ADD_REGISTRY_METADATA) 

128 await conn.execute(_ADD_REGISTRY_PROFILE) 

129 await conn.execute(_UPSERT_VERSION, _SCHEMA_VERSION) 

130 _schema_ready.add(id(pool))
