40async def load_sync_source(source: str, presets: PresetSource) -> SyncSource: 

41 """Fetch the named curated preset and parse it into the model list to merge. The 

42 fetch is the library's one networked operation, and it runs off the event loop.""" 

43 if not PRESET_NAME_RE.match(source): 

44 raise ValueError( 

45 f"unrecognized sync source {source!r} — a model list arrives as a curated preset" 

46 " name (e.g. 'freetier') and nothing else", 

47 ) 

48 text = await asyncio.to_thread(presets.text, source) 

49 return SyncSource(label=source, model_list=parse_model_list(tomllib.loads(text)))

55def _check_model_identity(arriving: list[LLMConfig], current: dict[str, LLMConfig]) -> None: 

56 for cfg in arriving: 

57 stored = current.get(cfg.name) 

58 if stored is not None and stored.model != cfg.model: 

59 raise ValueError( 

60 f"sync: refusing to change model for {cfg.name!r}" 

61 f" (stored {stored.model!r} vs preset {cfg.model!r}) — a model bump" 

62 " must be a new entry name", 

63 )

66def _check_name_clash(merged: list[LLMConfig]) -> None: 

67 """Refuse a name the merge itself would carry twice.""" 

68 seen: set[str] = set() 

69 for cfg in merged: 

70 if cfg.name in seen: 

71 raise ValueError( 

72 f"the merged model list would carry two entries named '{cfg.name}' —" 

73 " rename the entry this installation stated itself, the curated one" 

74 " carries the name the preset formed", 

75 ) 

76 seen.add(cfg.name)

79def _distinct_refs(entries: Iterable[LLMConfig]) -> tuple[str, ...]: 

80 return tuple(dict.fromkeys(c.api_key_ref for c in entries if c.api_key_ref))

83def _orphan_refs( 

84 removed: list[LLMConfig], 

85 merged: list[LLMConfig], 

86 present: frozenset[str], 

87) -> tuple[str, ...]: 

88 """Refs whose key exists here and which nothing in the merged model list references 

89 any more. A ref with no key behind it is nothing to revoke, and would be noise 

90 on the commonest removal of all.""" 

91 still_used = {c.api_key_ref for c in merged if c.api_key_ref} 

92 return tuple(ref for ref in _distinct_refs(removed) if ref not in still_used and ref in present)

95def _pending_keys( 

96 merged: list[LLMConfig], 

97 keys: dict[str, KeyInfo], 

98 present: frozenset[str], 

99) -> tuple[PendingKey, ...]: 

100 holds: dict[str, list[str]] = {} 

101 for cfg in merged: 

102 if not cfg.api_key_ref or cfg.api_key_ref in present: 

103 continue 

104 holds.setdefault(cfg.api_key_ref, []).append(cfg.name) 

105 return tuple( 

106 PendingKey( 

107 api_key_ref=ref, 

108 help=keys[ref].help if ref in keys else "", 

109 entry_names=tuple(names), 

110 ) 

111 for ref, names in holds.items() 

112 )

115def merge_upstream( 

116 new: ModelList, 

117 current: ModelList, 

118 present: frozenset[str], 

119 *, 

120 source: str, 

121) -> tuple[ModelList, SyncReport]: 

122 """Merge an arriving model list into the current one. Pure — no I/O, no secrets. 

123 

124 ``present`` names the refs a key resolves for; the report describes them and no 

125 decision here reads them. 

126 """ 

127 arriving = [c for c in new.configs if c.from_preset] 

128 stored_curated = [c for c in current.configs if c.from_preset] 

129 owned = [c for c in current.configs if not c.from_preset] 

130 current_by_name = {c.name: c for c in stored_curated} 

131 

132 _check_model_identity(arriving, current_by_name) 

133 

134 new_names = {c.name for c in arriving} 

135 removed = [c for c in stored_curated if c.name not in new_names] 

136 arrived = [c for c in arriving if c.name not in current_by_name] 

137 

138 merged = [*arriving, *owned] 

139 _check_name_clash(merged) 

140 

141 keys = dict(new.keys) 

142 for cfg in owned: 

143 if cfg.api_key_ref and cfg.api_key_ref not in keys and cfg.api_key_ref in current.keys: 

144 keys[cfg.api_key_ref] = current.keys[cfg.api_key_ref] 

145 

146 report = SyncReport( 

147 source=source, 

148 applied=True, 

149 added=tuple(c.name for c in arrived), 

150 updated=tuple( 

151 c.name for c in arriving if c.name in current_by_name and current_by_name[c.name] != c 

152 ), 

153 removed=tuple(c.name for c in removed), 

154 orphan_refs=_orphan_refs(removed, merged, present), 

155 pending_keys=_pending_keys(merged, keys, present), 

156 active_before=sum(1 for c in current.configs if c.api_key_ref in present), 

157 active_after=sum(1 for c in merged if c.api_key_ref in present), 

158 ) 

159 return ModelList(configs=merged, keys=keys), report

162def check_not_emptying( 

163 merged: list[LLMConfig], 

164 current: list[LLMConfig], 

165 report: SyncReport, 

166) -> None: 

167 """The one structural guard: never apply an empty model list over a working one. 

168 

169 An empty target accepts anything — that is onboarding, not a loss. 

170 """ 

171 if merged or not current: 

172 return 

173 raise SyncRefusedError( 

174 f"sync {report.source}: refusing to replace {len(current)} entries with an empty" 

175 " model list — nothing was changed", 

176 report=replace(report, applied=False), 

177 )

183def merge_model_list( 

184 src: SyncSource, 

185 current: ModelList, 

186 *, 

187 present: frozenset[str], 

188) -> MergeOutcome: 

189 """Everything above the writer: merge, and refuse an emptying. 

190 

191 Whoever holds the model_list, the decision is made here; only the write differs. 

192 """ 

193 merged, report = merge_upstream(src.model_list, current, present, source=src.label) 

194 check_not_emptying(merged.configs, current.configs, report) 

195 return MergeOutcome(report=report, model_list=merged)
